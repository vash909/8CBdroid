cmake_minimum_required(VERSION 3.22.1)
project(eightcbdroid_hamlib LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
    set(HAMLIB_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../build/hamlib/${ANDROID_ABI}")
elseif(NOT HAMLIB_PREFIX)
    message(FATAL_ERROR "Set HAMLIB_PREFIX to the verified host Hamlib build")
endif()
add_library(pinned_hamlib STATIC IMPORTED)
set_target_properties(pinned_hamlib PROPERTIES
    IMPORTED_LOCATION "${HAMLIB_PREFIX}/lib/libhamlib.a"
    INTERFACE_INCLUDE_DIRECTORIES "${HAMLIB_PREFIX}/include")
add_library(hamlib_adapter STATIC adapter.cpp)
set_target_properties(hamlib_adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(hamlib_adapter PUBLIC pinned_hamlib m)
if(ANDROID)
    add_library(eightcbdroid_hamlib SHARED jni.cpp)
    target_link_libraries(eightcbdroid_hamlib PRIVATE hamlib_adapter)
    target_link_options(eightcbdroid_hamlib PRIVATE "-Wl,--exclude-libs,ALL")
else()
    find_package(Threads REQUIRED)
    add_executable(hamlib_checks checks.cpp)
    target_link_libraries(hamlib_checks PRIVATE hamlib_adapter Threads::Threads)
    enable_testing()
    add_test(NAME hamlib_catalog_dummy COMMAND hamlib_checks)
    set_tests_properties(hamlib_catalog_dummy PROPERTIES TIMEOUT 20)
endif()
